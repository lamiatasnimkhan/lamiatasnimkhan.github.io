\section*{Selected Research Projects}

\project
{Multimodal AI for Early Detection of Parkinson's Disease}
{Python, PyTorch, Transformers, Medical AI}
{2025--Present}

\begin{itemize}
    \item Investigated multimodal learning approaches for early Parkinson's
    disease detection using heterogeneous clinical and biomedical data.
    \item Designed modality-specific deep learning encoders and
    Transformer-based multimodal fusion strategies.
    \item Explored SHAP and Grad-CAM for improving model interpretability.
\end{itemize}

\project
{Bengali Comic Conversation Understanding}
{Multimodal NLP, Vision-Language Models}
{2026--Present}

\begin{itemize}
    \item Developing a benchmark for evaluating vision-language models on
    Bengali comic conversation understanding.
    \item Constructing evaluation data from Bengali comics and question-answer
    pairs to assess multimodal reasoning and conversational understanding.
\end{itemize}

\project
{Deceptive Bangla SMS: Stylometric and Human Evaluation}
{NLP, Stylometry, Human Evaluation}
{2026--Present}

\begin{itemize}
    \item Investigating linguistic and stylometric characteristics of
    AI-generated deceptive Bangla SMS.
    \item Combining computational stylometry with human evaluation to examine
    differences between human- and machine-generated deceptive messages.
\end{itemize}

\project
{Causal Validation of Memory Failures in LLM Agents}
{LLM Agents, Evaluation, Causal Analysis}
{2026--Present}

\begin{itemize}
    \item Investigating causal validation and identifiability of memory
    failures in LLM-based agents.
    \item Studying failure modes associated with memory extraction, update,
    deletion, retrieval, filtering, and generation.
\end{itemize}
